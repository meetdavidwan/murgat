# EXAMPLE 1: Action Counting
# Question: How many times does the one-armed monster appear?
# Options: [A] 1, [B] 2, [C] 4, [D] 5
def execute_command(video: Video, options: List[str]) -> str:
    # 1. Find the events dynamically
    # Broad search or specific context searches
    times_doorway = video.find("one-armed monster runs out of doorway", modality="visual")
    times_tv = video.find("monster emerges after TV puzzle", modality="visual")
    times_attack = video.find("monster rushes player OR in sniper scope", modality="visual")

    # 2. Verify the findings
    sightings = []
    
    # Helper to query a list of times
    for t in times_doorway:
        sightings.append(video.query(t, "Check if the monster runs out of the doorway", modality="visual"))
    for t in times_tv:
        sightings.append(video.query(t, "Check if the monster emerges after the TV puzzle", modality="visual"))
    for t in times_attack:
        sightings.append(video.query(t, "Check for subsequent attacks by the monster", modality="visual"))

    evidence = {{
        "verified_sightings": sightings
    }}

    return answer_question(
        question="How many times does the one-armed monster appear?",
        evidence=evidence,
        options=options
    )

# EXAMPLE 2: Audio-Visual Grounding
# Question: Who in the video says, "This is called 'Song To You'"?
# Options: [A] The Drummer, [B] The Guitarist, [C] The Singer, [D] The Host
def execute_command(video: Video, options: List[str]) -> str:
    # 1. Find the specific audio event
    speech_segments = video.find("someone saying 'This is called Song To You'", modality="audio")

    # 2. Inspect the visual source for each segment found
    evidence_list = []
    
    for segment in speech_segments:
        # Verify the audio
        audio_obs = video.query(segment, "Transcribe the exact speech.", modality="audio")
        
        # Describe the speaker visually at that same timestamp
        visual_obs = video.query(segment, "Describe the appearance of the person speaking.", modality="visual")
        
        evidence_list.append({{"audio": audio_obs, "visual": visual_obs}})

    return answer_question(
        question="Who in the video says, 'This is called Song To You'?",
        evidence={{"candidates": evidence_list}},
        options=options
    )
